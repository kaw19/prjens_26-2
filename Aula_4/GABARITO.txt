# 1. Divisão em Treino (75% = 150 amostras) e Teste (25% = 50 amostras)
X_treino, X_teste, y_treino, y_teste = train_test_split(
    X_motor, 
    y_motor, 
    test_size=0.25, 
    random_state=42
)

# 2. Instanciação e Treinamento da Árvore de Decisão (Profundidade = 2)
clf_motor = tree.DecisionTreeClassifier(criterion='entropy', max_depth=2, random_state=42)
clf_motor.fit(X_treino, y_treino)

# 3. Avaliação de Acurácia no Conjunto de Teste
y_previsto = clf_motor.predict(X_teste)
print(f"Acurácia no Conjunto de Teste: {metrics.accuracy_score(y_teste, y_previsto):.2%}")
print("Matriz de Confusão (Teste):\n", metrics.confusion_matrix(y_teste, y_previsto))

fig, ax = plt.subplots(figsize=(14, 7))

    feature_names=nomes_sensores,
    class_names=['Normal (0)', 'Manutenção (1)'],
    filled=True,
    rounded=True,
    fontsize=10,
    ax=ax
)

    txt = txt.replace('entropy =', 'Entropia =')
    txt = txt.replace('samples =', 'Amostras no nó =')
    txt = txt.replace('value =', '[Normal, Falha] =')
    txt = txt.replace('class =', 'Diagnóstico =')
